% Functions
\newcommand{\stunden}[1]{\pgfmathparse{(int(floor(#1/3600)))/100}\pgfmathprintnumber[skip 0.=true, dec sep={}, fixed, fixed zerofill, precision=2]\pgfmathresult}
\newcommand{\minuten}[1]{\pgfmathparse{(int(floor((#1/60)-(floor(#1/3600)*60))))/100}\pgfmathprintnumber[skip 0.=true, dec sep={}, fixed, fixed zerofill, precision=2]\pgfmathresult}
\newcommand{\sekunden}[1]{\pgfmathparse{(int(floor(#1-(floor(#1/3600)*3600)-(floor((#1/60)-(floor(#1/3600)*60))*60))))/100}\pgfmathprintnumber[skip 0.=true, dec sep={}, fixed, fixed zerofill, precision=2]\pgfmathresult}

\pgfmathdeclarefunction{prozent}{2}{\pgfmathparse{(#1/#2)*100}}
\pgfmathdeclarefunction{restzeit}{2}{\pgfmathparse{(#2*60)-#1}}
