% The animate package normally advances exactly one frame per Acrobat timer
% callback. Delayed callbacks would therefore accumulate drift on slower
% systems. Keep animate's controls and frame-duration metadata, but anchor
% forward playback to elapsed wall-clock time. A separate one-second heartbeat
% checks the clock even when the next visual frame is 30 or 60 seconds away.
\makeatletter
\let\exam@timer@original@animate@jsinit\@anim@@jsinit
\gdef\@anim@@jsinit#1{%
	\exam@timer@original@animate@jsinit{#1}%
	if(typeof a#1_wallClockInitialized==='undefined'){%
		a#1_wallClockInitialized=true;%
		var a#1_wallTimes=new Array();%
		var a#1_wallStart=0;%
		var a#1_wallPausedElapsed=0;%
		var a#1_wallUserPaused=false;%
		var a#1_originalPlayRight=a#1_playRight;%
		var a#1_timerPauseButton=this.getField('examTimerPauseButton');%
		var a#1_timerResumeButton=this.getField('examTimerResumeButton');%
		examTimerSetRunningControls=function(running){%
			if(a#1_timerPauseButton&&a#1_timerResumeButton){%
				a#1_timerPauseButton.display=running?display.visible:display.hidden;%
				a#1_timerResumeButton.display=running?display.hidden:display.visible;%
				curdoc.dirty=false;%
			}%
		};%
		a#1_wallTimes[0]=0;%
		for(var i=0;i<\@anim@maxframe;i++){%
			var frameFps=typeof a#1_nFpsAt==='undefined'?a#1_fps:a#1_nFpsAt[i];%
			a#1_wallTimes[i+1]=a#1_wallTimes[i]+1000/frameFps;%
		}%
		a#1_playRight=function(){%
			if(!a#1_playing){%
				var resumeElapsed=a#1_wallUserPaused?a#1_wallPausedElapsed:a#1_wallTimes[a#1_idx];%
				a#1_wallStart=(new Date()).getTime()-resumeElapsed;%
			}%
			a#1_originalPlayRight();%
			a#1_wallUserPaused=false;%
			examTimerSetRunningControls(true);%
			var heartbeatInt;%
			try{heartbeatInt=app.setInterval('a#1_gotoNext()',1000);}catch(e){}%
			if(heartbeatInt){%
				try{app.clearInterval(a#1_int);}catch(e){}%
				a#1_int=heartbeatInt;%
			}%
		};%
		a#1_gotoNext=function(){%
			var elapsed=(new Date()).getTime()-a#1_wallStart+50;%
			var target=a#1_idx;%
			while(target<\@anim@maxframe&&a#1_wallTimes[target+1]<=elapsed){target++;}%
			if(target>=\@anim@maxframe){%
				a#1_stopLast();%
				a#1_wallPausedElapsed=a#1_wallTimes[\@anim@maxframe];%
				a#1_wallUserPaused=true;%
				examTimerSetRunningControls(false);%
			}else if(target>a#1_idx){%
				a#1_seekFrame(target);%
			}%
		};%
		examTimerPause=function(){%
			if(a#1_playing){%
				a#1_wallPausedElapsed=Math.max(0,Math.min(a#1_wallTimes[\@anim@maxframe],(new Date()).getTime()-a#1_wallStart));%
				a#1_wallUserPaused=true;%
				a#1_pause();%
				examTimerSetRunningControls(false);%
			}%
		};%
		examTimerResume=function(){%
			if(!a#1_playing&&a#1_wallPausedElapsed<a#1_wallTimes[\@anim@maxframe]){%
				a#1_wallUserPaused=true;%
				a#1_playFwd();%
			}%
		};%
		examTimerCorrectElapsed=function(){%
			var wasPlaying=a#1_playing;%
			var currentElapsed=wasPlaying?(new Date()).getTime()-a#1_wallStart:(a#1_wallUserPaused?a#1_wallPausedElapsed:a#1_wallTimes[a#1_idx]);%
			currentElapsed=Math.max(0,Math.min(a#1_wallTimes[\@anim@maxframe],currentElapsed));%
			if(wasPlaying){examTimerPause();}%
			var pad=function(value){return value<10?'0'+value:String(value);};%
			var formatTime=function(milliseconds){%
				var total=Math.floor(milliseconds/1000);%
				var wholeHours=Math.floor(total/3600);%
				var afterHours=total-wholeHours*3600;%
				var wholeMinutes=Math.floor(afterHours/60);%
				var wholeSeconds=afterHours-wholeMinutes*60;%
				return wholeHours+':'+pad(wholeMinutes)+':'+pad(wholeSeconds);%
			};%
			var maximumText=formatTime(a#1_wallTimes[\@anim@maxframe]);%
			while(true){%
				var answer=app.response({cQuestion:'\timerCorrectionQuestion'+' '+maximumText,cTitle:'\timerCorrectionTitle',cDefault:formatTime(currentElapsed)});%
				if(answer===null){%
					if(wasPlaying){examTimerResume();}%
					return;%
				}%
				var parts=String(answer).split(':');%
				if(parts.length===3&&parts[0]!==''&&parts[1]!==''&&parts[2]!==''){%
					var hours=Number(parts[0]);%
					var minutes=Number(parts[1]);%
					var seconds=Number(parts[2]);%
					var correctedSeconds=hours*3600+minutes*60+seconds;%
					if(isFinite(hours)&&isFinite(minutes)&&isFinite(seconds)&&hours>=0&&minutes>=0&&minutes<60&&seconds>=0&&seconds<60&&Math.floor(hours)===hours&&Math.floor(minutes)===minutes&&Math.floor(seconds)===seconds&&correctedSeconds*1000<=a#1_wallTimes[\@anim@maxframe]){%
						var correctedElapsed=correctedSeconds*1000;%
						var correctedFrame=0;%
						while(correctedFrame<\@anim@maxframe&&a#1_wallTimes[correctedFrame+1]<=correctedElapsed+50){correctedFrame++;}%
						a#1_seekFrame(correctedFrame);%
						a#1_wallPausedElapsed=correctedElapsed;%
						a#1_wallUserPaused=true;%
						a#1_isPaused=true;%
						if(wasPlaying&&correctedFrame<\@anim@maxframe){examTimerResume();}%
						return;%
					}%
				}%
				app.alert('\timerCorrectionInvalid'+' '+maximumText+'.');%
			}%
		};%
	}%
}
